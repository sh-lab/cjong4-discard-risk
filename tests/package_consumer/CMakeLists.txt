cmake_minimum_required(VERSION 3.16)
project(cj4dr_package_consumer LANGUAGES C CXX)
find_package(cjong4_discard_risk 0.2 CONFIG REQUIRED)
add_executable(consumer main.c)
target_link_libraries(consumer PRIVATE cjong4_discard_risk::cj4dr)
add_executable(consumer_cpp main.cpp)
target_compile_features(consumer_cpp PRIVATE cxx_std_11)
target_link_libraries(consumer_cpp PRIVATE cjong4_discard_risk::cj4dr)
enable_testing()
add_test(NAME consumer_c COMMAND consumer)
add_test(NAME consumer_cpp COMMAND consumer_cpp)
